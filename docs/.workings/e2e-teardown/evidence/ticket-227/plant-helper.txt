// Planted faults on e2e/close-app.ts, one at a time; each runs the helper's cases, then restores the file byte for byte.
// Playwright loads the helper's TypeScript directly, so no build is involved.
import { readFileSync, writeFileSync } from 'node:fs'
import { spawnSync } from 'node:child_process'
const file = 'e2e/close-app.ts'
const original = readFileSync(file)
const text = original.toString('utf8')
const variants = {
  'main-only kill': [["  process.kill(-pid, 'SIGKILL')", "  process.kill(pid, 'SIGKILL')"]],
  'no wait for the settled close': [["    await closed\n    return\n", "    return\n"], ["  // Playwright's close settles once the group's pipes have closed\n  await closed\n", ""]],
  'no report': [["  report(note)\n", ""]]
}
for (const [name, reps] of Object.entries(variants)) {
  let t = text
  for (const [a, b] of reps) {
    if (t.split(a).length !== 2) { console.error('PLANT: expected exactly one occurrence for', name, ':', a); process.exit(3) }
    t = t.replace(a, b)
  }
  try {
    writeFileSync(file, t)
    const r = spawnSync('pnpm', ['-s', 'exec', 'playwright', 'test', 'e2e/close-app.spec.ts'], { encoding: 'utf8' })
    const lines = (r.stdout + r.stderr).split('\n').filter((l) => /✓|✘|\d+ (passed|failed)/.test(l))
    console.log(`== ${name}: exit ${r.status}\n${lines.join('\n')}`)
  } finally {
    writeFileSync(file, original)
  }
}
console.log('RESTORED byte-identical:', Buffer.compare(readFileSync(file), original) === 0)
