----- mutant-run.sh
#!/bin/bash
# Usage: mutant-run.sh <worktree> <scratch dir> <mutation name>
# Builds a scratch tree whose scripts/ holds the worktree's test and a changed copy of its gate; src/ and the root's
# README*.md are real copies (the test copies them, and cpSync copies a link that is itself the source as a link);
# every other top-level entry is a link (docs/ included: the test copies docs/prototypes through it, which is safe,
# and makes docs/.workings an empty directory of its own). Then runs the test, unchanged, in the scratch tree.
set -e
W=$1; X=$2; NAME=$3
rm -rf "$X"; mkdir -p "$X/scripts"
for e in "$W"/* "$W"/.[!.]*; do n=$(basename "$e"); case "$n" in scripts|src|.git|README*.md) ;; *) ln -s "$e" "$X/$n";; esac; done
cp -R "$W/src" "$X/src"
cp "$W"/README*.md "$X/"
cp "$W/scripts/check-shared-ui.test.mjs" "$X/scripts/"
python3 "$(dirname "$0")/mutate.py" "$NAME" "$W/scripts/check-shared-ui.mjs" "$X/scripts/check-shared-ui.mjs"
echo "\$ diff <the gate> <the changed copy>   # mutation: $NAME"
diff "$W/scripts/check-shared-ui.mjs" "$X/scripts/check-shared-ui.mjs" || true
set +e
echo "\$ node --test <copy>/scripts/check-shared-ui.test.mjs (the test unchanged)"
node --test "$X/scripts/check-shared-ui.test.mjs" > "$X.out" 2>&1; echo "exit=$?"
grep -E "^(✖|ℹ (tests|pass|fail)) " "$X.out" | grep -v "^✖ failing tests"
rm -rf "$X" "$X.out"
