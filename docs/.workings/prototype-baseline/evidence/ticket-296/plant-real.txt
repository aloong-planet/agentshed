// Plant one Chinese comment line at the end of two real copies, run the gate, restore byte for byte in finally.
import { readFileSync, writeFileSync } from 'node:fs'
import { spawnSync } from 'node:child_process'
const CN = String.fromCodePoint(0x53d8, 0x5f02)
// Built from code points, so that this script holds no Chinese of its own
const HAN = new RegExp(`[${String.fromCodePoint(0x4e00)}-${String.fromCodePoint(0x9fff)}]`, 'g')
const files = { 'docs/prototypes/demo-harness.js': `// ${CN}\n`, 'docs/prototypes/index.html': `<!-- ${CN} -->\n` }
const saved = Object.fromEntries(Object.keys(files).map((f) => [f, readFileSync(f)]))
try {
  for (const [f, line] of Object.entries(files)) writeFileSync(f, Buffer.concat([saved[f], Buffer.from(line)]))
  for (const f of Object.keys(files)) console.log(`planted: ${f}, now ${readFileSync(f).length} bytes (was ${saved[f].length}), last line: ${readFileSync(f, 'utf8').trimEnd().split('\n').pop().replace(HAN, '<CJK>')}`)
  const r = spawnSync(process.execPath, ['scripts/check-lang.mjs'], { encoding: 'utf8' })
  console.log(`GATE_EXIT=${r.status}`)
  console.log((r.stderr + r.stdout).replace(HAN, '<CJK>'))
} finally {
  for (const f of Object.keys(files)) writeFileSync(f, saved[f])
  for (const f of Object.keys(files)) console.log(`restored: ${f}, identical: ${Buffer.compare(readFileSync(f), saved[f]) === 0}`)
}
