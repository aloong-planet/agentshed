#!/bin/bash
# Runs a command with N busy processes beside it, then stops them. Usage: stress.sh <n> <command...>
n="$1"; shift
pids=()
for i in $(seq 1 "$n"); do node -e 'for(;;){}' & pids+=($!); done
"$@"
code=$?
kill "${pids[@]}" 2>/dev/null
wait 2>/dev/null
exit $code
