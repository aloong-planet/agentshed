## The scan script
// Counts structural features of user question texts in the local session corpus (counts only, no content).
import zlib from 'node:zlib'
import { readdirSync, readFileSync, statSync } from 'node:fs'
import { join } from 'node:path'
import { homedir } from 'node:os'

function walk(dir, out) {
  let ents
  try { ents = readdirSync(dir, { withFileTypes: true }) } catch { return out }
  for (const e of ents) {
    const p = join(dir, e.name)
    if (e.isDirectory()) walk(p, out)
    else if (e.isFile()) out.push(p)
  }
  return out
}

const roots = [join(homedir(), '.claude', 'projects'), join(homedir(), '.codex', 'sessions')]
const files = roots.flatMap((r) => walk(r, []))
const jsonl = files.filter((f) => f.endsWith('.jsonl') || f.endsWith('.jsonl.zst'))
const other = files.filter((f) => !f.endsWith('.jsonl'))
const otherExt = {}
for (const f of other) { const m = f.match(/\.[^./]+$/); const k = m ? m[0] : '(none)'; otherExt[k] = (otherExt[k] ?? 0) + 1 }

const feats = {
  newline: /\n/, cr: /\r/, tab: /\t/, spaceRun: / {2,}/, zwj: /\u{200D}/, regional: /[\u{1F1E6}-\u{1F1FF}]/u,
  skinTone: /[\u{1F3FB}-\u{1F3FF}]/u, vs16: /\u{FE0F}/, combining: /(?![\u{FE00}-\u{FE0F}])\p{M}/u, astral: /[\u{10000}-\u{10FFFF}]/u,
  dottedI: /\u0130/, cjk: /[\u{3040}-\u{30FF}\u{3400}-\u{9FFF}\u{AC00}-\u{D7AF}]/, longOver200: /^[\s\S]{201,}$/
}
const counts = Object.fromEntries(Object.keys(feats).map((k) => [k, 0]))
// Within the 20 code points before the first occurrence of some later word: approximated by "a multi-code-point
// grapheme or whitespace run anywhere after the 20th code point" (where a window start could land)
let texts = 0, bytes = 0, badLines = 0, failed = 0
let graphemeSplitPossible = 0
const seg = new Intl.Segmenter(undefined, { granularity: 'grapheme' })
for (const f of jsonl) {
  let raw
  try { raw = f.endsWith('.zst') ? zlib.zstdDecompressSync(readFileSync(f)).toString('utf8') : readFileSync(f, 'utf8') } catch { failed++; continue }
  bytes += raw.length
  for (const line of raw.split('\n')) {
    if (!line.trim()) continue
    let o
    try { o = JSON.parse(line) } catch { badLines++; continue }
    const ts = []
    if (o.type === 'user' && o.message && o.message.role === 'user') {
      const c = o.message.content
      if (typeof c === 'string') ts.push(c)
      else if (Array.isArray(c)) for (const b of c) if (b && b.type === 'text' && typeof b.text === 'string') ts.push(b.text)
    } else if (o.type === 'event_msg' && o.payload && o.payload.type === 'user_message' && typeof o.payload.message === 'string') {
      ts.push(o.payload.message)
    }
    for (const t of ts) {
      texts++
      for (const [k, re] of Object.entries(feats)) if (re.test(t)) counts[k]++
      // A grapheme made of several code points somewhere past the 20th code point
      const cps = Array.from(t)
      if (cps.length > 21) {
        let n = 0
        for (const { segment } of seg.segment(t)) {
          const len = Array.from(segment).length
          if (n > 20 && len > 1 && segment !== '\r\n') { graphemeSplitPossible++; break }
          n += len
        }
      }
    }
  }
}
console.log(JSON.stringify({ jsonlFiles: jsonl.length, otherFiles: other.length, otherExt, charsRead: bytes, badLines, failed, userTexts: texts, counts, multiCodePointGraphemePast20: graphemeSplitPossible }, null, 1))
