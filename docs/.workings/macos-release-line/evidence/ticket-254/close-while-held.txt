// DIAG-254: throwaway, deleted before any commit. How long app.close() takes while the download is held at its
// first step (a 50 ms poll in the main process) against a download held after its last step (no timer).
import type {} from '../src/renderer/src/env'
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'
import { tmpdir } from 'node:os'
import { test, expect, _electron as electron } from '@playwright/test'

for (const held of [true, false]) {
  test(`DIAG-254 close while ${held ? 'held at the first step' : 'held after the last step'}`, async () => {
    const home = mkdtempSync(join(tmpdir(), 'agentshed-update-home-'))
    const userData = mkdtempSync(join(tmpdir(), 'agentshed-update-data-'))
    const script = join(home, 'update.json')
    writeFileSync(script, JSON.stringify({ check: 'found', version: '9.9.0', steps: [0, 35], stepMs: 50, download: 'hold', ...(held ? { holdFirstStep: true } : {}) }))
    const app = await electron.launch({
      args: ['.', `--user-data-dir=${userData}`],
      env: { ...(process.env as Record<string, string>), NODE_ENV: 'production', AGENTSHED_HOME_OVERRIDE: home, AGENTSHED_NO_FOREGROUND: '1', AGENTSHED_SYSTEM_LANGUAGES: 'en-US', AGENTSHED_UPDATE_SCRIPT: script }
    })
    const win = await app.firstWindow()
    await expect(win.getByTestId('update-slot')).toHaveAttribute('aria-valuenow', held ? '0' : '35')
    const t0 = Date.now()
    await app.close()
    console.log(`[DIAG-254] ${held ? 'held at first step' : 'held after last step'}: app.close() took ${Date.now() - t0}ms`)
    rmSync(home, { recursive: true, force: true })
    rmSync(userData, { recursive: true, force: true })
  })
}
