## plant-startup-toast.mjs
// Plant P4: a toast at startup (3.2 s, like any other), as a regression that announced something on launch would show.
// Replaces one exact occurrence, rebuilds, runs the given Playwright arguments, then restores the file byte for byte and
// rebuilds. Usage: node plant-startup-toast.mjs <playwright args...>
import { readFileSync, writeFileSync } from 'node:fs'
import { execSync, spawnSync } from 'node:child_process'
const file = 'src/renderer/src/Toast.tsx'
const original = readFileSync(file)
const text = original.toString('utf8')
const from = "    window.addEventListener('agentshed-toast', onToast)\n"
const to = "    window.addEventListener('agentshed-toast', onToast)\n    toast('ok', 'Planted startup notice')\n"
if (text.split(from).length !== 2) { console.error('PLANT: expected exactly one occurrence'); process.exit(3) }
let code = 0
try {
  writeFileSync(file, text.replace(from, to))
  console.log('PLANT applied:', readFileSync(file, 'utf8').split('\n').find((l) => l.includes('Planted startup notice'))?.trim())
  execSync('pnpm -s build', { stdio: 'ignore' })
  console.log('PLANT in the build (occurrences of the planted text):', execSync("grep -o 'Planted startup notice' out/renderer/assets/*.js | wc -l").toString().trim())
  const r = spawnSync('pnpm', ['-s', 'exec', 'playwright', 'test', ...process.argv.slice(2)], { encoding: 'utf8' })
  code = r.status ?? 1
  console.log(r.stdout, r.stderr)
} finally {
  writeFileSync(file, original)
  console.log('RESTORED byte-identical:', Buffer.compare(readFileSync(file), original) === 0)
  execSync('pnpm -s build', { stdio: 'ignore' })
}
console.log('PLANTED RUN EXIT', code)
