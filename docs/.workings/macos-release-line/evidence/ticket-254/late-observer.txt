## the throwaway case
// DIAG-254: throwaway feedback loop for #254, deleted before any commit. DIAG_VARIANT=old runs the case as it
// stands on main; =new runs it with the hold. DIAG_LATE_MS waits that long before the case's first read, as a
// window that shows after the download has moved on.
import type {} from '../src/renderer/src/env'
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'
import { tmpdir } from 'node:os'
import { test, expect, _electron as electron, type Page } from '@playwright/test'

const VARIANT = process.env.DIAG_VARIANT ?? 'old'
const LATE_MS = Number(process.env.DIAG_LATE_MS ?? '0')
const slot = (win: Page) => win.getByTestId('update-slot')
const tip = (win: Page) => win.getByTestId('update-tip')

test(`DIAG-254 ${VARIANT} case, first read ${LATE_MS}ms late`, async () => {
  const home = mkdtempSync(join(tmpdir(), 'agentshed-update-home-'))
  const userData = mkdtempSync(join(tmpdir(), 'agentshed-update-data-'))
  const script = join(home, 'update.json')
  const download = VARIANT === 'old'
    ? { check: 'found', version: '9.9.0', steps: [0, 20, 40, 60], stepMs: 700, download: 'hold' }
    : { check: 'found', version: '9.9.0', steps: [0, 20, 40, 60], stepMs: 100, download: 'hold' }
  writeFileSync(script, JSON.stringify(VARIANT === 'old' ? download : { ...download, holdFirstStep: true }))
  const app = await electron.launch({
    args: ['.', `--user-data-dir=${userData}`],
    env: { ...(process.env as Record<string, string>), NODE_ENV: 'production', AGENTSHED_HOME_OVERRIDE: home, AGENTSHED_NO_FOREGROUND: '1', AGENTSHED_SYSTEM_LANGUAGES: 'en-US', AGENTSHED_UPDATE_SCRIPT: script }
  })
  try {
    const win = await app.firstWindow()
    await win.waitForTimeout(LATE_MS)
    await expect(slot(win)).toHaveAttribute('aria-valuenow', '0')
    await slot(win).hover()
    await expect(tip(win)).toHaveText('Downloading update 0%')
    const where = async () => tip(win).evaluate((el) => {
      const r = (el.parentElement as HTMLElement).getBoundingClientRect()
      return { x: r.x, centre: r.y + r.height / 2 }
    })
    const first = await where()
    if (VARIANT !== 'old') writeFileSync(script, JSON.stringify(download))
    await expect(tip(win)).toHaveText('Downloading update 60%', { timeout: 5000 })
    expect(await where()).toEqual(first)
  } finally {
    await app.close()
    rmSync(home, { recursive: true, force: true })
    rmSync(userData, { recursive: true, force: true })
  }
})
