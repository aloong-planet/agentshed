# #254 code review, 2026-10-23, macOS: can a reader catch the script half-written while a test rewrites it?
# writer.mjs rewrites the script in a tight loop for 3 s (writeFileSync, two contents in turn); reader.mjs reads and parses it for 2.5 s
# command: node writer.mjs update.json & ; sleep 0.2; node reader.mjs update.json

## writer.mjs
import { writeFileSync } from 'node:fs'
const f = process.argv[2]
const a = JSON.stringify({ check: 'found', version: '9.9.0', steps: [0, 20, 40, 60], stepMs: 100, download: 'hold', holdFirstStep: true })
const b = JSON.stringify({ check: 'found', version: '9.9.0', steps: [0, 20, 40, 60], stepMs: 100, download: 'hold' })
const end = Date.now() + 3000
let n = 0
while (Date.now() < end) { writeFileSync(f, n++ % 2 ? a : b) }
console.log('writes', n)

## reader.mjs
import { readFileSync } from 'node:fs'
const f = process.argv[2]
const end = Date.now() + 2500
let reads = 0, failures = 0, empty = 0
while (Date.now() < end) {
  reads++
  let t = ''
  try { t = readFileSync(f, 'utf8'); JSON.parse(t) } catch { failures++; if (t === '') empty++ }
}
console.log('reads', reads, 'unparseable', failures, 'of which empty', empty)
