// DIAG-254: throwaway, deleted before any commit. Does a request the window makes after a click reach the main process
// after the click's own? Twenty times: "Check now" clicked with the check held, then getSettingsCheck at once; it must
// say checking each time if the click's request was handled first.
import type {} from '../src/renderer/src/env'
import { existsSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'
import { tmpdir } from 'node:os'
import { test, expect, _electron as electron } from '@playwright/test'

test('DIAG-254 request order from one window', async () => {
  const home = mkdtempSync(join(tmpdir(), 'agentshed-update-home-'))
  const userData = mkdtempSync(join(tmpdir(), 'agentshed-update-data-'))
  const script = join(home, 'update.json')
  writeFileSync(script, JSON.stringify({ check: 'upToDate' }))
  const app = await electron.launch({
    args: ['.', `--user-data-dir=${userData}`],
    env: { ...(process.env as Record<string, string>), NODE_ENV: 'production', AGENTSHED_HOME_OVERRIDE: home, AGENTSHED_NO_FOREGROUND: '1', AGENTSHED_SYSTEM_LANGUAGES: 'en-US', AGENTSHED_UPDATE_SCRIPT: script }
  })
  try {
    const win = await app.firstWindow()
    await win.locator('.ri.set').click()
    await expect.poll(() => existsSync(`${script}.log`) && readFileSync(`${script}.log`, 'utf8').length > 0).toBe(true)
    await expect.poll(() => win.evaluate(() => window.agentshed.getUpdateState().then((s) => s.phase))).toBe('idle')
    const seen: boolean[] = []
    for (let i = 0; i < 20; i++) {
      writeFileSync(script, JSON.stringify({ check: 'upToDate', holdCheck: true }))
      await win.getByTestId('version-action').click()
      seen.push((await win.evaluate(() => window.agentshed.getSettingsCheck())).checking)
      writeFileSync(script, JSON.stringify({ check: 'upToDate' }))
      await expect.poll(() => win.evaluate(() => window.agentshed.getSettingsCheck().then((c) => c.checking))).toBe(false)
    }
    console.log(`[DIAG-254] checking right after the click: ${seen.filter(Boolean).length} of ${seen.length}`)
  } finally {
    await app.close()
    rmSync(home, { recursive: true, force: true })
    rmSync(userData, { recursive: true, force: true })
  }
})
