// DIAG-254: throwaway feedback loop for #254, deleted before any commit
import type {} from '../src/renderer/src/env'
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'
import { tmpdir } from 'node:os'
import { test, expect, _electron as electron, type Page } from '@playwright/test'

const STEP_MS = Number(process.env.DIAG_STEP_MS ?? '700')
const slot = (win: Page) => win.getByTestId('update-slot')
const tip = (win: Page) => win.getByTestId('update-tip')

test('DIAG-254 the case as written, with the step time from DIAG_STEP_MS', async () => {
  const home = mkdtempSync(join(tmpdir(), 'agentshed-update-home-'))
  const userData = mkdtempSync(join(tmpdir(), 'agentshed-update-data-'))
  const script = join(home, 'update.json')
  writeFileSync(script, JSON.stringify({ check: 'found', version: '9.9.0', steps: [0, 20, 40, 60], stepMs: STEP_MS, download: 'hold' }))
  const t0 = Date.now()
  const app = await electron.launch({
    args: ['.', `--user-data-dir=${userData}`],
    env: { ...(process.env as Record<string, string>), NODE_ENV: 'production', AGENTSHED_HOME_OVERRIDE: home, AGENTSHED_NO_FOREGROUND: '1', AGENTSHED_SYSTEM_LANGUAGES: 'en-US', AGENTSHED_UPDATE_SCRIPT: script }
  })
  try {
    const win = await app.firstWindow()
    await slot(win).waitFor()
    console.log(`[DIAG-254] stepMs=${STEP_MS} slot visible after ${Date.now() - t0}ms reading ${await slot(win).getAttribute('aria-valuenow')}`)
    await expect(slot(win)).toHaveAttribute('aria-valuenow', '0')
    await slot(win).hover()
    await expect(tip(win)).toHaveText('Downloading update 0%')
    await expect(tip(win)).toHaveText('Downloading update 60%', { timeout: 5000 })
  } finally {
    await app.close()
    rmSync(home, { recursive: true, force: true })
    rmSync(userData, { recursive: true, force: true })
  }
})
