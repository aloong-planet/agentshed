## the throwaway case
// DIAG-254: throwaway feedback loop for the failed-download case, deleted before any commit. DIAG_VARIANT=old runs
// its second launch as on main; =new with the hold. DIAG_LATE_MS waits that long before the case's first read.
import type {} from '../src/renderer/src/env'
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'
import { tmpdir } from 'node:os'
import { test, expect, _electron as electron } from '@playwright/test'

const VARIANT = process.env.DIAG_VARIANT ?? 'old'
const LATE_MS = Number(process.env.DIAG_LATE_MS ?? '0')

test(`DIAG-254 failed download, ${VARIANT} case, first read ${LATE_MS}ms late`, async () => {
  const home = mkdtempSync(join(tmpdir(), 'agentshed-update-home-'))
  const userData = mkdtempSync(join(tmpdir(), 'agentshed-update-data-'))
  const script = join(home, 'update.json')
  const download = VARIANT === 'old'
    ? { check: 'found', version: '9.9.0', steps: [0, 30], stepMs: 400, download: 'fail' }
    : { check: 'found', version: '9.9.0', steps: [0, 30], stepMs: 100, download: 'fail' }
  writeFileSync(script, JSON.stringify(VARIANT === 'old' ? download : { ...download, holdFirstStep: true }))
  const app = await electron.launch({
    args: ['.', `--user-data-dir=${userData}`],
    env: { ...(process.env as Record<string, string>), NODE_ENV: 'production', AGENTSHED_HOME_OVERRIDE: home, AGENTSHED_NO_FOREGROUND: '1', AGENTSHED_SYSTEM_LANGUAGES: 'en-US', AGENTSHED_UPDATE_SCRIPT: script }
  })
  try {
    const win = await app.firstWindow()
    await win.waitForTimeout(LATE_MS)
    await expect(win.getByTestId('update-slot')).toBeVisible()
    if (VARIANT !== 'old') writeFileSync(script, JSON.stringify(download))
    await expect(win.getByTestId('update-slot')).toHaveCount(0, { timeout: 5000 })
    await expect(win.locator('.toast')).toHaveCount(0)
  } finally {
    await app.close()
    rmSync(home, { recursive: true, force: true })
    rmSync(userData, { recursive: true, force: true })
  }
})
