## plant-notice.mjs
// Plant P2: a failed background download sends a notice at once (the install failure's notice), as a regression of
// app-update::REQ-002/AC-02 would. Replaces one exact occurrence, rebuilds, runs the given Playwright arguments, then
// restores the file byte for byte and rebuilds. Usage: node plant-notice.mjs <playwright args...>
import { readFileSync, writeFileSync } from 'node:fs'
import { execSync, spawnSync } from 'node:child_process'
const file = 'src/main/update-controller.ts'
const original = readFileSync(file)
const text = original.toString('utf8')
const from = `        // Codex's lifecycle has no failure state: the button goes and the next check downloads again
        () => this.set(IDLE_UPDATE)`
const to = `        // Codex's lifecycle has no failure state: the button goes and the next check downloads again
        () => { this.set(IDLE_UPDATE); this.deps.notify({ code: 'installFailed' }) }`
if (text.split(from).length !== 2) { console.error('PLANT: expected exactly one occurrence'); process.exit(3) }
let code = 0
try {
  writeFileSync(file, text.replace(from, to))
  console.log('PLANT applied:', readFileSync(file, 'utf8').split('\n').find((l) => l.includes("notify({ code: 'installFailed' }) }"))?.trim())
  execSync('pnpm -s build', { stdio: 'ignore' })
  console.log('PLANT in the build (lines of out/main with the planted notify after a set):', execSync(`grep -c "IDLE_UPDATE);" out/main/index.js || true`).toString().trim())
  const r = spawnSync('pnpm', ['-s', 'exec', 'playwright', 'test', ...process.argv.slice(2)], { encoding: 'utf8' })
  code = r.status ?? 1
  console.log(r.stdout, r.stderr)
} finally {
  writeFileSync(file, original)
  console.log('RESTORED byte-identical:', Buffer.compare(readFileSync(file), original) === 0)
  execSync('pnpm -s build', { stdio: 'ignore' })
}
console.log('PLANTED RUN EXIT', code)
