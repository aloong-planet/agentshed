## the throwaway case
// DIAG-299: throwaway feedback loop for #299, deleted before any commit. The startup-restore case's rollover step
// alone: today's card shows 2k, the page's clock is installed at 00:30 and fast-forwarded past a minute, and the card
// must read 0. DIAG_RATE throttles the page's CPU (CDP), DIAG_TIMEOUT is the step's window (the case uses 1000).
import type {} from '../src/renderer/src/env'
import { mkdtempSync, mkdirSync, writeFileSync, rmSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { test, expect, _electron as electron } from '@playwright/test'

const RATE = Number(process.env.DIAG_RATE ?? '1')
const TIMEOUT = Number(process.env.DIAG_TIMEOUT ?? '1000')

test(`DIAG-299 rollover, CPU x${RATE}, window ${TIMEOUT}ms`, async () => {
  const home = mkdtempSync(join(tmpdir(), 'agentshed-restore-home-'))
  const userData = mkdtempSync(join(tmpdir(), 'agentshed-restore-data-'))
  const proj = join(home, 'restore-project')
  mkdirSync(proj)
  writeFileSync(join(home, '.claude.json'), JSON.stringify({ projects: { [proj]: {} } }))
  const cdir = join(home, '.claude', 'projects', proj.replace(/[^a-zA-Z0-9]/g, '-'))
  mkdirSync(cdir, { recursive: true })
  const timestamp = new Date().toISOString()
  writeFileSync(join(cdir, 'saved.jsonl'), [
    { type: 'user', timestamp, message: { role: 'user', content: 'Saved question' } },
    { type: 'assistant', timestamp, message: { model: 'claude-fable-5', content: [{ type: 'text', text: 'Saved answer' }], usage: { input_tokens: 1200, output_tokens: 300, cache_read_input_tokens: 0, cache_creation_input_tokens: 0 } } }
  ].map(x => JSON.stringify(x)).join('\n') + '\n')
  const app = await electron.launch({ args: ['.', `--user-data-dir=${userData}`], env: {
    ...process.env, NODE_ENV: 'production', AGENTSHED_HOME_OVERRIDE: home, AGENTSHED_NO_FOREGROUND: '1', AGENTSHED_SYSTEM_LANGUAGES: 'en-US'
  } })
  try {
    const win = await app.firstWindow()
    await win.locator('.rail .ri').nth(0).click()
    await expect(win.locator('.tot-c').nth(1)).toContainText('2k')
    const cdp = await win.context().newCDPSession(win)
    if (RATE > 1) await cdp.send('Emulation.setCPUThrottlingRate', { rate: RATE })
    await new Promise((r) => setTimeout(r, Number(process.env.DIAG_PRE_MS ?? '0')))
    await win.clock.install({ time: new Date('2030-01-01T00:30:00Z') })
    await new Promise((r) => setTimeout(r, Number(process.env.DIAG_MID_MS ?? '0')))
    await win.clock.fastForward(61_000)
    const t0 = Date.now()
    if (process.env.DIAG_HOLD) {
      // Hold the page's main thread through the debugger from 600 ms after the fast-forward for 700 ms, as a
      // loaded machine would: the display clock's tick, due about 900 ms after, runs late
      await cdp.send('Debugger.enable')
      setTimeout(() => {
        void cdp.send('Debugger.pause').then(() => new Promise((r) => setTimeout(r, 700))).then(() => cdp.send('Debugger.resume')).catch(() => {})
      }, 600)
    }
    if (process.env.DIAG_POLL) {
      // Node-side polling every 10 ms (real timers), independent of how Playwright's assertion retries
      const card = win.locator('.tot-c').nth(1).locator('.v')
      let text = await card.textContent()
      while (text !== '0' && Date.now() - t0 < 10_000) { await new Promise((r) => setTimeout(r, 10)); text = await card.textContent() }
      console.log(`[DIAG-299] node poll: card read ${text} ${Date.now() - t0}ms after the fast-forward`)
    }
    try {
      await expect(win.locator('.tot-c').nth(1).locator('.v')).toHaveText('0', { timeout: TIMEOUT })
    } finally {
      console.log(`[DIAG-299] CPU x${RATE}: card read ${await win.locator('.tot-c').nth(1).locator('.v').textContent()} ${Date.now() - t0}ms after the fast-forward`)
    }
    await cdp.detach()
  } finally {
    await app.close()
    rmSync(home, { recursive: true, force: true })
    rmSync(userData, { recursive: true, force: true })
  }
})
