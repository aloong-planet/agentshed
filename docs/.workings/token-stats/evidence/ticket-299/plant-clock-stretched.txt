import { execSync, spawnSync } from 'node:child_process'

const original = readFileSync(file)
const text = original.toString('utf8')
const from = 'const timer = setInterval(tick, 1000)'
const to = 'const timer = setInterval(tick, 60_000)'
if (text.split(from).length !== 2) { console.error('PLANT: expected exactly one occurrence'); process.exit(3) }
let code = 0
try {

  console.log('PLANT applied:', readFileSync(file, 'utf8').split('\n').find((l) => l.includes('60_000'))?.trim())

  console.log(r.stdout, r.stderr)
} finally {
  writeFileSync(file, original)
  console.log('RESTORED byte-identical:', Buffer.compare(readFileSync(file), original) === 0)
  execSync('pnpm -s build', { stdio: 'ignore' })
}
